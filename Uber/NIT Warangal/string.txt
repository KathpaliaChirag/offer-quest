#include <iostream>
using namespace std;

string solution(string s){
    int n = s.size();
    
    for(int i=0; i<n; i++){
        if(s[i] == '?'){
            if(i<n-2 && s[i+1] == s[i+2] && s[i+1] != '?'){
                if(s[i+1] != '?' && s[i+1] =='a'){
                    s[i] = 'b';
                }
                else if(s[i+1] != '?' && s[i+1] =='b'){
                    s[i] = 'a';
                }
                else{
                    s[i] = 'b';
                }
            }
            else if(i>=1 && i<n-1 && s[i-1] == s[i+1]){
                if(s[i-1] == 'a'){
                    s[i] = 'b';
                }
                else
                    s[i] = 'a';
            }
            else if(i >=2 && s[i-1] == s[i-2] ){
                if(s[i-1] == 'a')
                    s[i] = 'b';
                else
                    s[i] = 'a';
            }
            else
                s[i] = 'a';
        }
    }
    return s;
}

int main() {
    string s = "aa??aa";
    cout<<solution(s);
}
