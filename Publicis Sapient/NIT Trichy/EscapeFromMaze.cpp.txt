#include<bits/stdc++.h>
using namespace std;

vector<long> ans;
int N;
vector<bool> visited;

void fun(vector<vector<int>>& edges,vector<int> edgelength , vector<long> t, int i, int time1){
    
    if(visited[i])
        return;
        
    if(t[i] > time1 && ans[i] > time1)
        ans[i] = time1;
    else if(t[i] <= time1)
        return;
    
    visited[i] = true;
    
    for(int j=0; j<N; j++){
        if(edges[i][j] != -1){
            fun(edges, edgelength, t, j, time1+edgelength[edges[i][j]]);
        }
    }
    visited[i] = false;
}

vector<long> mazeGame(vector<long> t, vector<int> endpoint1, vector<int> endpoint2,vector<int> edgelength){
    
    int n = t.size();
    int m = edgelength.size();
    N =n;
    
    ans.resize(n,INT_MAX);
    
    visited.resize(n,false);
    
    vector<vector<int>> edges(n,vector<int>(n,-1));
    
    //mark the edges
    for(int i=0; i<m; i++){
        int f = endpoint1[i]-1;
        int s = endpoint2[i]-1;
        edges[f][s] = i;
        edges[s][f] = i;
    }
    
    fun(edges,edgelength, t, 0 ,0);
    
    for(int i =0; i<m; i++){
        if(ans[i] == INT_MAX)
            ans[i] = -1;
    }
    
    return ans;
    
}

int main(){
    int n =4, m = 4;
    vector<int> endpoint1 = {1,2,3,4};
    vector<int> endpoint2 = {2,4,1,3};
    vector<int> edgelength = {2,1,5,3};
    vector<long> t = {1,4,7,9};
    
    vector<long> ans= mazeGame(t, endpoint1, endpoint2, edgelength);
    
    for(int i=0; i<ans.size(); i++)
        cout<<ans[i]<<" ";
    
    
    return 0;
    
}
