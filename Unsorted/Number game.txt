//1ST CODE

#include <bits/stdc++.h>
using namespace std;

int MaxDistinct(vector<int> nums){
    unordered_map<int,int> map;
    
    for(auto i : nums)
        map[i]++;
    
    unordered_map<int, int>:: iterator p;
    int count1=0;
    int count2=0;
    for (p = map.begin(); p != map.end(); p++){
        if(p->second %2 == 0)
            count2++;
        count1++;
    }
    if(count2 % 2 )
        count1--;
    
    return count1;
}

int main() {
	vector<int> v = {1,2,2,3,3,4,4,3,3,3,3,3,3};
	int k =0;
	cout<<MaxDistinct(v);
	return 0;
}
