class Solution{
public:
    string longestPalindrome(string s) {
        int n = s.size();
        vector<vector<int>> dp(n,vector<int>(n,1));
        int i1 = 0;
        int j1 = 0;
        for(int k=1; k<n; k++){
            for(int l=0; l<n-k; l++){
                int i = l;
                int j = k+l;
               if(dp[i+1][j-1] == 1 && s[i] == s[j]){
                   dp[i][j] = 1;
                   if(j-i > j1-i1){
                       i1 = i;
                       j1 = j;
                   }
               }else
                   dp[i][j] = 0;
            }
        }
        return s.substr(i1,j1-i1+1);               
    }
};
