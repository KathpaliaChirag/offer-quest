#include<bits/stdc++.h>
using namespace std;

long getMaxUnit(int num, vector<int> boxes, int unitSize, vector<int> unitPerBox, long truckSize){
    
    priority_queue<pair<int,int>> pq;
    
    for(int i=0; i<num; i++)
        pq.push({unitPerBox[i], boxes[i]});
    
    long count = 0;
    int k = 0;
    while(!pq.empty()){
        pair<int,int> x = pq.top();
        pq.pop();
        //check if truck is full and break;
        if(k+x.second >= truckSize){
            count += (x.first)*(truckSize - k);
            break;
        }
        else{
            k += x.second;
            count += (x.first)*(x.second);
        }
    }
    return count;
}

int main(){
    vector<int> boxes = {5,2,4,3};
    vector<int> unitPerBox = {10,5,7,9};
    cout<<getMaxUnit(4,boxes,4,unitPerBox,10);
    	return 0;
}
