#include <iostream>
#include<bits/stdc++.h>
using namespace std;

class node{
    public:
    node* left;
    node* right;
    
    node(){
      left = NULL;
      right = NULL;
    }
};

class trie{
  node* root;
  public:
    trie(){
        root = new node();
    }

    void insert(int data){
      node* temp = root;
      for(int i=31; i>=0; i--){
          
          int bit = 1&(data>>i);
          //if bit is 0 we go in left
          if(bit == 0){
              if(temp->left == NULL){
                temp->left = new node();
              }
              temp = temp->left;
          }
          else{
              if(temp->right == NULL){
                temp->right = new node();
              }
              temp = temp->right;
          }
      }
    }
    int max_xor(int data){
      if(!root->left && !root->right) return 0;
        node* temp = root;
        int max_xor_val = 0;
        for(int i = 31; i >= 0; i--){
            int bit = (data >> i) & 1;
            if(bit == 0){
                if(temp->right != NULL){ 
                    max_xor_val += (1 << i);
                    temp = temp->right;
                }
                else temp = temp->left;
            }
            else{
                if(temp->left != NULL){
                    max_xor_val += (1 << i);
                    temp = temp->left;
                }
                else temp = temp->right;
            }
        }
        return max_xor_val;
    }
};

int solve(int* a, int* b, int n){
    trie* A = new trie();
    trie* B = new trie();
    int count=0;
    for(int i=0; i<n; i++){
      int x = a[i];
      int y = b[i];

      x = max(x, A->max_xor(x));
      y = max(y, B->max_xor(y));

      if(x > y){
        B->insert(b[i]);
        count++;
      }
      else if(x < y){
        A->insert(a[i]);
      }
    }
    return count;
}


int main() {
  int n =4;
  int a[10] = {10,1,14,3,1,3,14,4,1,13};
  int b[10] = {14,14,11,12,8,15,16,16,18,3};
  
  cout<<solve(a,b,10)<<endl;
}
