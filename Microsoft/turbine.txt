int solution( vector<vector<int>> &A, int k)
{
    int m = A.size();
    int n = A[0].size();
    
    print(A);
    
    queue<pair<int,int>> bird;
    
    for( int i = 0; i<m ; i++ )
    {
        for( int j =0; j<n ; j++)
        {
            if(A[i][j]==1)
                bird.push({i,j});
        }
    }
    
    if(bird.size()==0)
        return m*n;
    if(bird.size()==m*n)
        return 0;
    
    while(bird.size())
    {
        pair<int,int> temp = bird.front();
        bird.pop();
        queue<pair<int,int>> q;
        q.push(temp);
        
        while (!q.empty()) 
        {
            pair<int,int> p = q.front();
            q.pop();
            
            int i = p.first;
            int j = p.second;
       
            if (i < 0 || i >= m || j < 0 || j >= n)
                continue;
       
            if ( (A[i][j] == 1 && i!=temp.first && j!=temp.second) || A[i][j]==-1)
                continue;

       
            A[i][j] = -1;
 
            for (int x = -1; x <= 1; x += 2) 
            {
                if( abs(i+x-temp.first) + abs(j-temp.second) <= k)
                    q.push({i + x, j});
                if( abs(i-temp.first) + abs(j+x-temp.second) <= k)
                    q.push({i, j + x});
            }
        }
    
        
    }
    
    //print(A);
    
    int count = 0;
    
    for( int i = 0; i<m ; i++ )
    {
        for( int j =0; j<n ; j++)
        {
            if(A[i][j]==0)
                count++;
        }
    }
    
    return count;
    
}

int main()
{
   int m,n;
   cin>>m>>n;
   int k;
   cin>>k;
   
   vector<vector<int>> A;
   
   for(int i = 1;i<=m ; i++)
   {
       vector<int> temp;
       int t;
       for( int j = 1; j<=n; j++)
       {
           cin>>t;
           temp.push_back(t);
       }
       A.push_back(temp);
   }
  
   cout<<solution(A,k);
   
   return 0;
}
